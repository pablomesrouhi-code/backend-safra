## 7.3 Post-CTA Upsell Selection Logic

```typescript
function selectUpsell(cartSkus: string[]): Product | null {
  const ALL = ['freshguard', 'heatshield', 'underguard'];
  const missing = ALL.filter(id => !cartSkus.includes(id));
  if (missing.length === 0) return null; // 3 products → skip modal

  const AFFINITY: Record<string, string> = {
    freshguard: 'underguard',  // breath → underarm (social proximity)
    heatshield: 'underguard',  // body sweat → underarm
    underguard: 'freshguard',   // underarm → breath (full confidence)
  };

  if (cartSkus.length === 1) {
    return AFFINITY[cartSkus[0]] ?? missing[0];
  }
  // 2 in cart → upsell the 3rd missing
  return missing[0];
}
```
